cmake_minimum_required(VERSION 3.22.1)
project(odaiko_native LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_library(odaiko SHARED OdaikoCore.c OdaikoProfile.cpp TaikoNative.cpp)
set_source_files_properties(OdaikoCore.c PROPERTIES COMPILE_OPTIONS "-fno-builtin")
target_compile_options(odaiko PRIVATE -O2 -Wall -Wextra -Wno-unused-parameter)
find_package(oboe REQUIRED CONFIG)
target_link_libraries(odaiko PRIVATE oboe::oboe android log m)
target_link_options(odaiko PRIVATE "-Wl,-z,max-page-size=16384")
